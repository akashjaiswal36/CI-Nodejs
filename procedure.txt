🧩 What You Will Build (High Level)

A production-style DevOps platform for a NodeJS (NestJS-like) backend + React frontend, deployed on EKS, using:

GitHub Actions (CI)

GitOps (Helm + ArgoCD)

AWS (ECR, IAM, S3, Secrets Manager, CloudWatch)

Canary deployments

Security scanning + SBOM

Observability

SLOs & runbooks

🔐 GitHub Actions CI (Very Important)

📄 .github/workflows/ci.yml

name: CI Pipeline

on:
  push:
    branches: [ main ]

jobs:
  build:
    runs-on: ubuntu-latest
    permissions:
      id-token: write
      contents: read

    steps:
    - uses: actions/checkout@v4

    - name: Configure AWS Credentials
      uses: aws-actions/configure-aws-credentials@v4
      with:
        role-to-assume: arn:aws:iam::123456789012:role/github-ci-role
        aws-region: ap-south-1

    - name: Build Docker Image
      run: docker build -t nodejs-app:${{ github.sha }} .

    - name: Push to ECR
      run: |
        aws ecr get-login-password | docker login --username AWS --password-stdin 123456789012.dkr.ecr.ap-south-1.amazonaws.com
        docker tag nodejs-app:${{ github.sha }} 123456789012.dkr.ecr.ap-south-1.amazonaws.com/nodejs-app:${{ github.sha }}
        docker push 123456789012.dkr.ecr.ap-south-1.amazonaws.com/nodejs-app:${{ github.sha }}

📄 src/routes/ui.js

const router = require('express').Router();
const os = require('os');

router.get('/', (req, res) => {
  res.send(`
    <html>
      <head>
        <title>DevOps Platform Demo</title>
        <style>
          body { font-family: Arial; background: #f4f6f8; padding: 40px; }
          .card { background: white; padding: 20px; border-radius: 8px; width: 420px; }
          h2 { color: #2c3e50; }
        </style>
      </head>
      <body>
        <div class="card">
          <h2>🚀 Siprahub DevOps Platform Demo</h2>
          <p><b>Status:</b> UP</p>
          <p><b>Version:</b> ${process.env.APP_VERSION || 'v1'}</p>
          <p><b>Environment:</b> ${process.env.ENV || 'dev'}</p>
          <p><b>Pod:</b> ${os.hostname()}</p>
          <p><b>Timestamp:</b> ${new Date().toISOString()}</p>
        </div>
      </body>
    </html>
  `);
});

module.exports = router;
